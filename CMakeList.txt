cmake_minimum_required(VERSION 3.20)

project(
    market-model
    VERSION 0.1.0
    DESCRIPTION "C++ market simulation model"
    LANGUAGES CXX
)

# ==== C++ standard ===================================

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(BUILD_TESTS "Build tests" ON)
option(BUILD_EXAMPLES "Build examples" ON)

# ==== Library ========================================

add_library(market_model

    # ---- Market ----
    
    src/market/market.cpp
    src/market/matching_engine.cpp
    src/market/order_book.cpp
    src/market/order_registry.cpp
    src/market/trade_history.cpp

)

target_include_directories(
    market_model
    PUBLIC
        ${PROJECT_SOURCE_DIR}/include
)

target_compile_features(
    market_model
    PUBLIC
        cxx_std_20
)

# ==== Compiler Warnings =============================

if (MSVC)

    target_compile_options(
        market_model
        PRIVATE
            /W4
    )

else ()

    target_compile_options(
        market_model
        PRIVATE
            -Wall
            -Wextra
            -Wpedantic
    )

endif ()

# ==== Examples ======================================

if (BUILD_EXAMPLES)

    add_exacutable(
        market_model_demo
        examples/main.cpp
    )

    target_link_libraries(
        market_model_demo
        PRIVATE
            market_model
    )

endif ()

# ==== Tests =========================================

if (BUILD_TESTS)

    enable_testing()

    # ---- Order Lifecycle Test ----

    add_exacutable(
        order_lifecycle_test
        tests/market/order_lifecycle_test.cpp
    )

    target_link_libraries(
        order_lifecycle_test
        PRIVATE
            market_model
    )

    add_test(
        NAME order_lifecycle_test
        COMMAND order_lifecycle_test
    )

    # ---- Order Registry Test ----

    add_exacutable(
        order_registry_test
        tests/market/order_registry_test.cpp
    )

    target_link_libraries(
        order_registry_test
        PRIVATE
            market_model
    )

    add_test(
        NAME order_registry_test
        COMMAND order_registry_test
    )

    # ---- Trade History Test ----

    add_exacutable(
        trade_history_test
        tests/market/trade_history_test.cpp
    )

    target_link_libraries(
        trade_history_test
        PRIVATE
            market_model
    )

    add_test(
        NAME trade_history_test
        COMMAND trade_history_test
    )

endif ()
